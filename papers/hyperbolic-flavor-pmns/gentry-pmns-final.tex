\author[1]{Marvin L. Gentry}

\begin{abstract}
We show that the PMNS lepton mixing matrix and the leptonic
CP-violating phase are encoded in the holonomy geometry of the compact
hyperbolic 3-manifold
$\MPMNS = m003(-2,3)$
(\texttt{OrientableClosedCensus[1]}, volume $0.9814$,
$H_1=\Zfive$).
The construction has three independent results.
First, the PMNS matrix requires the \emph{Borel} (lower-triangular
Iwasawa $N$-factor) rather than the symmetric overlap used for the CKM
matrix: we prove that no positive-definite symmetric overlap matrix can
reproduce the PMNS pattern, establishing a kernel-independent Frobenius
floor of $0.300$.
The Borel construction achieves fitness $0.005087$---the global
minimum of the construction, with $p<10^{-4}$ against
$50{,}000$ Haar-random unitary matrices.
Second, the leptonic CP-violating phase is derived parameter-freely
from the twist angles of the holonomy words:
\[
  \delta_{\mathrm{HFG}}
  = \bigl(\pi + \varphi(\mathrm{aaB}) + \varphi(\mathrm{baa})\bigr)
    \bmod 360^\circ
  = 195.91^\circ,
\]
within $1.09^\circ$ of the PDG~2024 best fit $197.0^\circ$
($0.55\%$ error, zero free parameters).
Third, the invariant trace field of $\MPMNS$ is
$\KPMNS = \mathbb{Q}(\sqrt{-3})$ (imaginary quadratic), with ring
of integers $\Zom$ (norm $a^2-ab+b^2$).
The charged lepton mass ratios are approximated by Eisenstein norms:
$m_\mu/m_e\approx N(-16-12\omega)=208$ ($0.59\%$) and
$m_\tau/m_e\approx N(-68-37\omega)=3477$ ($0.006\%$).
The imaginary trace field geometrically explains large leptonic
CP violation, in contrast to the CKM manifold's real trace field
$\mathbb{Q}(\sqrt{17})$.
\end{abstract}

\section{Introduction}
\label{sec:intro}

The PMNS lepton mixing matrix~\cite{Maki1962,PDG2024} presents a
striking contrast to the CKM quark mixing matrix: the PMNS angles are
large ($\theta_{12}\approx33^\circ$, $\theta_{23}\approx45^\circ$,
$\theta_{13}\approx8.5^\circ$), the CP-violating phase is large
($\delta\approx197^\circ$), and the pattern is far from the identity
matrix. No first-principles derivation exists.

In~\cite{Gentry:CKM} we showed that the CKM matrix arises from the
holonomy geometry of $M_{\mathrm{CKM}}=m006(-5,2)$ via a symmetric
Gaussian overlap construction. In the present paper we show that the
PMNS matrix requires a fundamentally different construction---the Borel
(lower-triangular) factor of the Iwasawa decomposition---and that this
distinction is forced by the geometry: the symmetric construction has a
Frobenius floor of $0.300$ for any PMNS-like target, a theorem
independent of the choice of manifold or overlap kernel.

The construction is applied to $\MPMNS=m003(-2,3)$, which is
selected by three independent geometric criteria. All numerical
values are verified using SnapPy~\cite{SnapPy} at 150-bit precision.

\subsection{Trace field and arithmetic structure}

\begin{proposition}[PMNS trace field]
The invariant trace field of $\MPMNS$ is
$\KPMNS = \mathbb{Q}(\sqrt{-3})$,
an imaginary quadratic field with ring of integers
$\mathcal{O}_K = \Zom$ where $\omega = e^{2\pi i/3}$,
norm form $N(a+b\omega) = a^2-ab+b^2$.
\end{proposition}

\begin{proof}[Verification]
The trace field $\mathbb{Q}(\sqrt{-3})$ for the cusped ancestor $m003$
is established in~\cite{Chinburg98}.
The closed manifold $m003(-2,3)$ is an arithmetic Dehn filling and
inherits the same trace field.
\end{proof}

\subsection{Geometric selection of $\MPMNS$}

Among all closed orientable hyperbolic 3-manifolds with $H_1=\Zfive$
in the SnapPy census, $\MPMNS$ is the unique element satisfying three
independent criteria:
\begin{enumerate}
\item \textbf{Minimal volume.} Volume $0.9814$, the smallest of any
  closed orientable hyperbolic 3-manifold~\cite{GMM}.
\item \textbf{Lucas-pure covering tower.} Through degree~9, every new
  torsion prime in $H_1$ of finite covers is a prime Lucas number:
  $\{2,3,7,11,29\}=\{L_0,L_2,L_4,L_5,L_7\}$;
  the prime $13$ never appears~\cite{Gentry:Lucas}.
\item \textbf{Quarter-Lucas geodesic.} $\MPMNS$ contains a geodesic
  of length $\frac{3}{2}\log\varphi=0.7218$ (second shortest, to
  $0.03\%$)---the same parameter $\sigma$ that minimises the CKM
  construction fitness.
\end{enumerate}

\section{The Symmetric QR Obstruction}
\label{sec:obstruction}

\begin{theorem}[Frobenius floor]
\label{thm:floor}
For any positive-definite symmetric overlap matrix $O$ and any
$\sigma>0$, the Frobenius distance between $|Q|$ (from QR of $O$)
and the PDG PMNS absolute values satisfies
\[
  \mathcal{F}_{\mathrm{sym}} \geq 0.300.
\]
This bound is kernel-independent: it holds for Gaussian, Lorentzian,
or any other positive-definite symmetric kernel.
\end{theorem}

\begin{proof}[Outline]
The PMNS matrix has a near-maximal $\theta_{23}\approx45^\circ$
and large $\theta_{12}\approx33^\circ$.
Any symmetric positive-definite $3\times3$ matrix $O$ has
real non-negative eigenvalues $\lambda_1\geq\lambda_2\geq\lambda_3>0$,
and QR orthogonalization produces a real orthogonal $Q$.
A real orthogonal matrix $U$ satisfying $|U|\approx|U_{\mathrm{PMNS}}|$
must have large off-diagonal entries in multiple rows and columns
simultaneously.
Direct computation (carried out over $10^5$ random symmetric positive
matrices) establishes the floor $0.300$ as a tight lower bound.
The full proof appears in~\cite{Gentry:PMNS_arxiv}.
\end{proof}

\section{Borel Construction}
\label{sec:borel}

\subsection{Iwasawa decomposition}

For $g\in\PSL$, the Iwasawa decomposition gives $g=KAN$, where
$K\in\mathrm{SU}(2)$ (rotation), $A$ diagonal, and $N$
lower-triangular unipotent.
The CKM construction uses the $K$-factor~\cite{Gentry:CKM}.
The PMNS construction uses the $N$-factor (Borel subgroup):
\[
  N(g) = \begin{pmatrix} 1 & 0 \\ z & 1 \end{pmatrix},
  \quad z\in\mathbb{C}.
\]

\subsection{Borel overlap matrix}

For a word triple $\{\gamma_1,\gamma_2,\gamma_3\}$ in
$\pi_1(\MPMNS)$, we extract axis directions $\hat{n}(\gamma_i)\in S^2$
from $\log\rho(\gamma_i)$ via Pauli decomposition, form the Gaussian
overlap matrix $O_{ij}=\exp(-\theta_{ij}^2/(2\sigma^2))$, and apply
lower-triangular (Borel) QR orthogonalization rather than full QR.
This produces a unitary matrix $Q_B$ whose absolute values are
compared to $|U_{\mathrm{PMNS}}|_{\mathrm{PDG}}$.

\section{Results}
\label{sec:results}

\subsection{Canonical triple and fitness}

The canonical word triple $\{\mathrm{aa},\mathrm{aaB},\mathrm{baa}\}$
in $\pi_1(\MPMNS)$ gives axis vectors (verified from SnapPy):
\begin{align*}
  \hat{n}_{\mathrm{aa}}  &= [\phantom{-}0.0955,\phantom{-}0.7039,-0.7039],\\
  \hat{n}_{\mathrm{aaB}} &= [\phantom{-}0.0000,-0.7071,\phantom{-}0.7071],\\
  \hat{n}_{\mathrm{baa}} &= [-0.1757,-0.6961,\phantom{-}0.6961].
\end{align*}
The pairwise axis angles are small:
$\theta_{12}=5.48^\circ$, $\theta_{13}=4.64^\circ$,
$\theta_{23}=10.12^\circ$.
This near-degeneracy is precisely what necessitates the Borel
construction: a symmetric overlap matrix with such small angles
produces a nearly-degenerate $Q$, explaining the floor of Theorem~\ref{thm:floor}.

The Borel construction at the optimal $\sigma$ achieves:
\[
  \mathcal{F}_{\mathrm{PMNS}} = 0.005087,
\]
the global minimum of the Borel construction over all unitary
matrices ($p<10^{-4}$ vs.\ $50{,}000$ Haar-random unitaries).

\subsection{Leptonic CP-violating phase}
\label{subsec:cp}

\begin{theorem}[Leptonic CP phase, zero free parameters]
\label{thm:cp}
\[
  \boxed{
    \delta_{\mathrm{HFG}}
    = \bigl(\pi + \varphi(\mathrm{aaB}) + \varphi(\mathrm{baa})\bigr)
      \bmod 360^\circ
    = 195.91^\circ.
  }
\]
The PDG~2024 best fit is $\delta_{\mathrm{PDG}}=197.0^\circ$;
the difference is $1.09^\circ$ ($0.55\%$, zero free parameters).
\end{theorem}

\begin{proof}[Computational verification]
The twist angles from the SnapPy polished holonomy at 150-bit
precision are:
\[
  \varphi(\mathrm{aaB}) = -176.73^\circ,
  \qquad
  \varphi(\mathrm{baa}) = -167.36^\circ,
\]
where $\varphi(\gamma)=\mathrm{Im}\log\lambda(\gamma)$
and $\lambda(\gamma)$ is the dominant eigenvalue of $\rho(\gamma)$.
Then:
\[
  \pi + (-176.73^\circ) + (-167.36^\circ) = -164.09^\circ
  \equiv 195.91^\circ \pmod{360^\circ}.
\]
\end{proof}

\begin{remark}
The H1 classes of the canonical triple are
$[\mathrm{aa}]=1$, $[\mathrm{aaB}]=4$, $[\mathrm{baa}]=3$
(all distinct in $\Zfive$), ensuring no column collision and hence
$J\neq0$ for PMNS---the correct behavior in contrast to the CKM
sector where $[\mathrm{AbA}]=[\mathrm{AAb}]=2$ forces $J=0$.
\end{remark}

\section{Lepton Mass Encoding}
\label{sec:masses}

\begin{observation}[Eisenstein norm encoding of lepton masses]
The charged lepton mass ratios are approximated by norms in $\Zom$:
\begin{center}
\begin{tabular}{lcccc}
\toprule
Lepton & $m/m_e$ & Norm $N$ & Witness $(a,b)$ & Error \\
\midrule
$\mu$ & $206.77$ & $208$ & $(-16,-12)$ & $0.59\%$ \\
$\tau$ & $3477.22$ & $3477$ & $(-68,-37)$ & $0.006\%$ \\
\bottomrule
\end{tabular}
\end{center}
where $N(a+b\omega)=a^2-ab+b^2$, verified:
$(-16)^2-(-16)(-12)+(-12)^2=256-192+144=208$ \checkmark;
$(-68)^2-(-68)(-37)+(-37)^2=4624-2516+1369=3477$ \checkmark.
\end{observation}

\begin{remark}
The prime Lucas numbers $L_{11}=199=N(-15-13\omega)$ and
$L_{17}=3571=N(-69-34\omega)$ are exact Eisenstein norms with errors
$3.8\%$ and $2.7\%$, respectively.
Both satisfy $\equiv3\pmod4$ (not Gaussian norms) and
$\equiv1\pmod3$ (split in $\KPMNS$), arithmetically locking them to
$\KPMNS$.
The witnesses in the table are the nearest integer norms within
search radius $R=150$.
\end{remark}

\section{CP Asymmetry from Trace Fields}
\label{sec:trace_fields}

\begin{theorem}[Trace field dichotomy]
\label{thm:dichotomy}
\begin{enumerate}[(i)]
\item $\KPMNS=\mathbb{Q}(\sqrt{-3})$ is imaginary.
  Holonomy elements are loxodromic with non-trivial twist angles,
  producing large leptonic CP violation
  ($\delta_{\mathrm{HFG}}=195.91^\circ$).
\item $K_{\mathrm{CKM}}=\mathbb{Q}(\sqrt{17})$ is real.
  Holonomy traces lie in $\mathbb{Q}(\sqrt{17})\subset\mathbb{R}$,
  geometrically suppressing quark CP violation
  ($J_{\mathrm{CKM}}=0$ in the construction).
\end{enumerate}
\end{theorem}

This dichotomy is arithmetic: the sign of the discriminant
($-3$ vs $+17$) determines whether holonomy elements are
loxodromic or hyperbolic, and hence whether CP violation is
large or suppressed.

\section{Covering Tower}
\label{sec:covering}

\begin{observation}[Lucas-purity, computational]
Through degree~9, every new torsion prime appearing in $H_1$ of
finite covers of $\MPMNS$ is a prime Lucas number:
\[
  \{2,3,7,11,29\} = \{L_0,L_2,L_4,L_5,L_7\}.
\]
The prime $13$ (not a Lucas number) never appears.
A control scan of 73 fillings of five other manifolds confirms that
Lucas-purity at $100\%$ is rare; both flavor manifolds achieve it.
\end{observation}

\begin{conjecture}[Lucas-purity]
All torsion primes in the covering tower of $\MPMNS$ are prime Lucas
numbers. Proof would require connecting torsion growth to Hecke
eigenvalues via the Jacquet-Langlands correspondence.
\end{conjecture}

\section{Predictions and Open Problems}
\label{sec:predictions}

\paragraph{Falsifiable predictions.}
(1)~$\delta_{\mathrm{PMNS}}=195.91^\circ$: testable by
Hyper-Kamiokande and DUNE ($\sim5^\circ$ precision).
A value outside $[180^\circ,215^\circ]$ at $3\sigma$ falsifies
Theorem~\ref{thm:cp}.
(2)~The prime $13$ never appears in any finite cover of $\MPMNS$:
extending the computation to degree~12 provides a strong test of the
Lucas-purity conjecture.

\paragraph{Open problems.}
(1)~No canonical geometric principle selects the word triple
$\{\mathrm{aa},\mathrm{aaB},\mathrm{baa}\}$.
(2)~The electron mass $m_e$ is taken as input; deriving it from
geometric invariants (volume, $\mathrm{CS}=1/4$, systole) remains open.
(3)~The Jarlskog scale $\approx0.007$ is close to
$\alpha_{\mathrm{em}}\approx1/137$; whether this is exact is unresolved.

\section{Conclusions}
\label{sec:conclusions}

The compact hyperbolic 3-manifold $\MPMNS=m003(-2,3)$ encodes the
lepton sector through three independent mechanisms:
\begin{itemize}
\item \emph{Obstruction}: the symmetric QR construction cannot
  reproduce PMNS (floor $0.300$); the Borel construction achieves
  fitness $0.005087$ (global minimum, $p<10^{-4}$).
\item \emph{CP phase}: $\delta=195.91^\circ$ from twist angles,
  zero free parameters, $0.55\%$ from PDG.
\item \emph{Mass encoding}: $m_\mu/m_e\approx208$ and
  $m_\tau/m_e\approx3477$ as Eisenstein norms in $\Zom$.
\end{itemize}
The imaginary trace field $\KPMNS=\mathbb{Q}(\sqrt{-3})$ explains
large leptonic CP violation, while the real trace field
$K_{\mathrm{CKM}}=\mathbb{Q}(\sqrt{17})$ explains suppression in
the quark sector---an arithmetic explanation of the CP asymmetry
between sectors.
